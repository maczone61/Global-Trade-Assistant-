Please wait.' :'Searching for general apparel importers, wholesalers, distributors and stocklot buyers. Please wait.'; summary.innerHTML=''; results.innerHTML=''; try{ const outcome=await runBuyerSearch(product,countries,false); discoveredResults=outcome.results||[]; discoveredRejectedResults=outcome.rejectedResults||[]; const data=outcome.data||{}; const qualifiedCount=discoveredResults.filter(isQualifiedBuyerAssessment).length; const rejectedCount=discoveredRejectedResults.length; const assessmentErrors=data.assessment_errors??0; const searchErrors=data.search_errors||[]; const errorDetails=data.assessment_error_details||[]; let errorHtml=''; if(errorDetails.length){ errorHtml=`

Assessment error details

${errorDetails.map((e,i)=>`
Error ${i+1}: ${esc(e.company||'Unknown company')}

${esc(e.website||'')}

${esc(e.error||'Unknown error')}

`).join('')}
`; }else if(assessmentErrors>0){ errorHtml=`

The Worker reported assessment failures but did not return individual error messages.

`; } summary.innerHTML=`

Search report

Searches performed: ${esc(data.searches_performed??'Not supplied')}

Unique search results: ${esc(data.candidates_found??discoveredResults.length+rejectedCount)}

Companies assessed successfully: ${esc(data.candidates_assessed??'Not supplied')}

Assessment attempts: ${esc(data.assessment_attempted??'Not supplied')}

Assessment errors: ${esc(assessmentErrors)}

Search errors: ${esc(JSON.stringify(searchErrors))}

Qualified prospects shown: ${qualifiedCount}

Rejected candidates (not saved): ${rejectedCount}

New qualified prospects saved to your buyer register: ${esc(outcome.saved??0)}

${esc(data.warning||'Search results are prospects, not confirmed buyers.')}

${errorHtml}`; await refresh(); renderDiscoveryResults(); status.textContent=assessmentErrors>0 ?'Search completed, but some companies could not be assessed. See the error details above.' :qualifiedCount ?'Research complete. '+qualifiedCount+' qualified prospect(s) are shown.' :'Search completed.
